ADB Commands

# Enable developer settings
adb shell settings put global development_settings_enabled 1

# Enable ADB over WiFi
adb shell settings put global adb_wifi_enabled 1

# Set USB config to ADB
adb shell setprop persist.sys.usb.config adb

# Open USB mode chooser
adb shell am start -n com.android.settings/.connecteddevice.usb.UsbModeChooserActivity

# Open developer options
adb shell am start -a android.settings.APPLICATION_DEVELOPMENT_SETTINGS

# Open FQC factory test
adb shell am start -n com.longcheertel.cit/.CitSampleActivity
